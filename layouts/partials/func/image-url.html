{{- /* URL of an image with a Netlify CDN preset (e.g. "post-small", "hero-lg"). Without the CDN it returns the original path. Param: src, preset. */ -}} {{- $src := strings.TrimPrefix "/" .src -}} {{- if partial "func/cdn-enabled.html" . -}} {{- return printf "/cdn-%s/%s" .preset $src -}} {{- end -}} {{- return printf "/%s" $src -}}
