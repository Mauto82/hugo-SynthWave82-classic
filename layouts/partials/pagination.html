{{- /* Numbered pagination: 1 … (current-1) current (current+1) … last, with optional arrows. Param: the paginator (.Paginator or the result of .Paginate). */ -}} {{- $p := . -}} {{- if gt $p.TotalPages 1 -}} {{- $current := $p.PageNumber -}} {{- $last := $p.TotalPages -}} {{- $arrows := site.Params.home.showPaginatorArrows -}}
{{- if $arrows }} {{- with $p.Prev }} < {{- else }} < {{- end }} {{- end }} {{- range $p.Pagers }} {{- $n := .PageNumber -}} {{- $near := and (ge $n (sub $current 1)) (le $n (add $current 1)) -}} {{- if eq $n $current }} {{ $n }} {{- else if or (eq $n 1) (eq $n $last) $near }} {{ $n }} {{- end }} {{- /* dots after the first page and before the last when there's a gap */ -}} {{- if and (eq $n 1) (gt $current 3) }} ... {{- end }} {{- if and (eq $n (sub $last 1)) (lt $current (sub $last 2)) }} ... {{- end }} {{- end }} {{- if $arrows }} {{- with $p.Next }} > {{- else }} > {{- end }} {{- end }}
{{- end -}}
